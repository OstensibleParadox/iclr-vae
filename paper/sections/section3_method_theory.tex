\section{Certificates and Task-Aware Budgeting}
\label{sec:method-theory}

\subsection{The finite and asymptotic guarantees}

Write the covariance residue as
\begin{equation}
  \mathcal L_N^{\mathrm{ren}}(K_N)
  :=\frac12\sum_{j\le n_N}
    [\lambda_{j,N}-\log(1+\lambda_{j,N})].
  \label{eq:renormalized-objective}
\end{equation}
Its finite part is represented by $-\frac12\log\det_2(I+K_N)$
(\cref{app:det2}). The following result turns
$\mathfrak C_N=(\Ttwo(N),M_N,\delta_N^{\mathrm{cert}})$ into a decision
certificate.

\begin{theorem}[Finite raw-$\Ttwo$ certificate]
\label{thm:finite-certificate}
Let $K_N=K_N^\ast$, $I+K_N\succeq\delta I$ for $0<\delta\le1$, and
$\opnorm{K_N}\le M$. Then
\begin{equation}
  \frac{\Ttwo(N)}{4(1+M)^2}
  \le \mathcal L_N^{\mathrm{ren}}(K_N)
  \le \frac{\Ttwo(N)}{4\delta^2}.
  \label{eq:kl-t2-equivalence}
\end{equation}
If $X\sim q_N$ and
$Y_N=\log\frac{dq_N}{dp_N}(X)
-\E_{q_N}\log\frac{dq_N}{dp_N}(X)$, then
\begin{equation}
  \Var(Y_N)=\frac12\Ttwo(N),\qquad
  \Pr\!\left(
    |Y_N|\ge\sqrt{\Ttwo(N)x}+Mx
  \right)\le2e^{-x}.
  \label{eq:log-density-variance}
\end{equation}
For $K_N\succeq0$, the objective constants sharpen to
$[4(1+M)]^{-1}$ and $1/4$.
\end{theorem}

Thus $\Ttwo$ is an exact variance budget but not, by itself, a complete tail
certificate: $M$ and a chosen failure probability control rare fluctuations,
while $\delta$ controls proximity to the covariance boundary. Proofs and the
explicit confidence-radius form are in
\cref{app:finite-certificate-proof}.

\begin{theorem}[Uniform refinement threshold]
\label{thm:s2-stability}
For a refinement family with common $(\delta,M)$,
\begin{equation}
  \sup_N\Ttwo(N)<\infty
  \quad\Longleftrightarrow\quad
  \sup_N\mathcal L_N^{\mathrm{ren}}(K_N)<\infty
  \quad\Longleftrightarrow\quad
  \sup_N\Var(Y_N)<\infty.
  \label{eq:s2-threshold}
\end{equation}
If $\Ttwo(N)\to\infty$, the other two quantities diverge.
\end{theorem}
Uniform boundedness is a no-divergence statement, not convergence by itself.
Compatible $\Schatten{2}$ convergence and the familiar
$s_j\asymp j^{-\alpha}$ transition at $\alpha=1/2$ are stated precisely in
\cref{app:continuum-realization}.

\subsection{Allocate the covariance budget}
\label{sec:task-aware-budget}

Let $B_N$ be a noise-amplitude operator in whitened coordinates. Its relative
covariance is
\begin{equation}
  K_N=B_NB_N^\ast,\qquad
  \Delta C_N=C_N^{1/2}K_NC_N^{1/2}.
  \label{eq:amplitude-covariance-contract}
\end{equation}
Hence a diagonal amplitude $b_j$ contributes $b_j^4$ to raw $\Ttwo$. Throughout,
$\tau$ denotes the \emph{squared} budget $\hsnorm{K_N}^2\le\tau$.
The basis-invariant unweighted baseline is radial projection,
\begin{equation}
  \Phi_\tau^{\mathrm{rad}}(K_N)
  =\gamma_NK_N,\qquad
  \gamma_N=\min\!\left\{1,\sqrt{\frac{\tau}{\Ttwo(N)}}\right\},
  \label{eq:s2-budget-gate}
\end{equation}

with the convention $\gamma_N=1$ when $K_N=0$. Our task-aware rule fixes a
refinement-compatible basis, assumes the positive
injected covariance is diagonal in that basis, and uses nonnegative entries
$q_{j,N}$ from a local PSD task model (e.g., squared sensitivities or a
Gauss--Newton diagonal):
\begin{equation}
  \min_{0\le\widetilde\lambda_{j,N}\le\rho_{\mathrm{cap}}}
  \frac12\sum_jq_{j,N}
    (\widetilde\lambda_{j,N}-\lambda_{j,N})^2
  \quad\text{s.t.}\quad
  \sum_j\widetilde\lambda_{j,N}^2\le\tau.
  \label{eq:task-aware-projection}
\end{equation}
Its KKT solution is
\begin{equation}
  \widetilde\lambda_{j,N}
  =\min\!\left\{\rho_{\mathrm{cap}},
    \frac{q_{j,N}}{q_{j,N}+\mu}\lambda_{j,N}\right\},
  \qquad \mu\ge0,
  \label{eq:task-aware-kkt}
\end{equation}
where $\rho_{\mathrm{cap}}>0$, bisection chooses $\mu$ when the budget is
active, and the canonical value is zero if $q_{j,N}=\mu=0$. Larger task weight
means less departure from the target perturbation, independent of physical
frequency. This is a closed-form commuting approximation, not a solution to
general nonconvex task distortion; raw signed gradients are not valid weights.
The derivation and modewise envelope variant are in
\cref{app:task-aware-kkt}.

To implement the covariance solution with an amplitude factor, let
$K_N=U_N\operatorname{diag}(\lambda_{j,N})U_N^\ast$ and set
\begin{equation}
  g_{j,N}
  =\sqrt{\frac{\widetilde\lambda_{j,N}}{\lambda_{j,N}}},
  \qquad
  \widetilde B_N
  =U_N\operatorname{diag}(g_{j,N})U_N^\ast B_N,
  \label{eq:covariance-to-amplitude-gate}
\end{equation}
with $g_{j,N}=1$ when both eigenvalues are zero. Then
$\widetilde B_N\widetilde B_N^\ast
=U_N\operatorname{diag}(\widetilde\lambda_{j,N})U_N^\ast$; using covariance
ratios directly as amplitude gains would square the intended shrinkage.

\subsection{Resolution and diffusion time}
\label{sec:temporal-budgets}

Let $K_{N,t}(X_{t-1})$ be the conditional relative covariance update between two
Markov transitions, whitened by the conditional reference. We keep four
quantities distinct:
\begin{align}
  \mathcal B_{\max}(N,T)
    &:=\max_{1\le t\le T}\E_q\Ttwo(K_{N,t}),\label{eq:max-t2-budget}\\
  \mathcal B_{\mathrm{path}}(N,T)
    &:=\sum_{t=1}^T\E_q\Ttwo(K_{N,t}),\label{eq:path-t2-budget}\\
  \mathcal B_{\mathrm{train}}(N,T)
    &:=\sum_{t=1}^Tw_t\E_{\pi_t}\Ttwo(K_{N,t}),\label{eq:training-t2-budget}\\
  \mathcal B_{\mathrm{end}}(N,T)
    &:=\hsnorm{
      (C^p_{N,T})^{-1/2}(C^q_{N,T}-C^p_{N,T})(C^p_{N,T})^{-1/2}
    }^2.\label{eq:endpoint-t2-budget}
\end{align}
Under common certificate constants and $q_0=p_0$,
$\mathcal B_{\mathrm{path}}$ controls the conditional-transition covariance
part of path KL and equals twice the variance of the centered covariance
log-ratio martingale. An initial KL and conditional mean shifts, when present,
are added separately. The weighted
training quantity uses the estimator distribution $\pi_t$ and is not path KL
unless $w_t=1$ and $\pi_t$ is the path marginal. The endpoint follows the
actual covariance recursion and is generally neither a sum nor a product of
stepwise updates. Derivations and the weighted variance identity are in
\cref{app:temporal-budgets}.

\subsection{Latent implementation and audit}

For encoder Jacobian $J_N(x)$, input perturbation covariance $\Sigma_N$, and a
fixed latent reference $C_{z,N}^{\mathrm{ref}}$, define
\begin{equation}
  F_N(x)=(C_{z,N}^{\mathrm{ref}})^{-1/2}J_N(x)\Sigma_N^{1/2},
  \qquad K_{E,N}(x)=F_N(x)F_N(x)^\ast.
  \label{eq:encoder-pushforward}
\end{equation}
The mean-map component has the exact identity
\begin{equation}
  \Ttwo^{(E)}(N;x)
  =\hsnorm{K_{E,N}(x)}^2
  =\sum_js_j(F_N(x))^4
  =\lVert F_N(x)\rVert_{\Schatten{4}}^4.
  \label{eq:encoder-s4-identity}
\end{equation}
For a stochastic VAE, let $S_{\mathrm{post},N}(x)$ be its conditional latent
posterior covariance and take $C_{z,N}^{\mathrm{ref}}$ to be the clean
encoder-mean covariance. Under the local, independent, zero-mean input
perturbation model, write $W_N=(C_{z,N}^{\mathrm{ref}})^{-1/2}$. The local and
population injected covariances are
\begin{equation}
  \begin{aligned}
    K_{E,N}^{\mathrm{full}}(x)
      &:=K_{E,N}(x)+W_NS_{\mathrm{post},N}(x)W_N^\ast,\\
    \overline K_{E,N}^{\mathrm{full}}
      &:=\E_x K_{E,N}^{\mathrm{full}}(x),&
    \Ttwo^{(E,\mathrm{full})}(N)
      &:=\hsnorm{\overline K_{E,N}^{\mathrm{full}}}^2.
  \end{aligned}
  \label{eq:full-encoder-covariance}
\end{equation}
The pointwise $K_{E,N}^{\mathrm{full}}(x)$ defines a local
covariance-matched Gaussian proxy certificate. Its population average is generally a
non-Gaussian mixture: $\Ttwo^{(E,\mathrm{full})}$ is then a covariance budget,
not a log-density tail certificate. We report it separately from the
mean-map-only $\Ttwo^{(E)}(N;x)$.

The implementable adapter is sample independent. We freeze one nested
Fourier--channel basis across resolutions, apply the task-aware KKT rule to the
diagonal structured component of $\overline K_{E,N}^{\mathrm{full}}$, and use
the square-root amplitude gains in
\cref{eq:covariance-to-amplitude-gate}. Because the implicit full operator need
not commute with this basis, a final radial rescaling certifies
\begin{equation}
  \widetilde K_{E,N}^{\mathrm{full}}
  =G_{\psi,N}\overline K_{E,N}^{\mathrm{full}}G_{\psi,N}^\ast,
  \qquad
  \hsnorm{\widetilde K_{E,N}^{\mathrm{full}}}^2\le\tau_E
  \quad\text{for every audited }N.
  \label{eq:fixed-whitened-latent-gate}
\end{equation}
The guarantee is for the declared population covariance, not every input
pointwise. The gate changes only stochastic residual amplitudes in the fixed
whitening geometry. An invertible adapter that transforms both representation
and reference would instead preserve the singular values of $F_N$
(\cref{app:encoder-audit}).

We audit the encoder pushforward separately from the downstream
\emph{marginal} diffusion diagnostic
\begin{equation}
  K^{\mathrm{marg}}_{\mathrm{diff},N,t}
  =(C_{z,N}^{\mathrm{ref}})^{-1/2}
    (C_{z,N,t}-C_{z,N}^{\mathrm{ref}})
   (C_{z,N}^{\mathrm{ref}})^{-1/2}.
  \label{eq:diffusion-relative-covariance}
\end{equation}
This object compares time-$t$ marginal covariances; it is not the conditional
transition update used in $\mathcal B_{\mathrm{path}}$.
Structured operators are summed exactly. For an implicit real operator, fixed
independent Rademacher probes~\citep{hutchinson1989stochastic} give
\begin{equation}
  \widehat\Ttwo_m(N)
  =\frac1m\sum_{r=1}^m\lVert K_Nv_r\rVert_2^2,
  \qquad \E\widehat\Ttwo_m(N)=\Ttwo(N).
  \label{eq:matrix-free-t2}
\end{equation}
Training uses a separate cheap surrogate; final probes are independent of
training, fixed across paired comparisons, and reported with operator calls,
time, and memory. Probe variance, stopping rules, complex operators, and the
difference between global certificates and blockwise diagnostics are detailed
in \cref{app:estimators}.
